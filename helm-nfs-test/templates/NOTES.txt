   kubectl get pvc {{ .Values.storage.pvcName }}

   kubectl get pv

   - NodePort Access: http://<NODE-IP>:{{ .Values.service.nodePort }}
   - Port Forwarding Access:
     kubectl port-forward svc/{{ include "nfs-storage-test.fullname" . }} 8080:80
